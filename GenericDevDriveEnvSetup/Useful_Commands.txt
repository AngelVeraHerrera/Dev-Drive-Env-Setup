# =====================================================================
# DEV DRIVE MANAGEMENT EXAMPLE COMMANDS
# =====================================================================

--- ALLOW POWERSHELL SCRIPT EXECUTION ---
Set-ExecutionPolicy RemoteSigned -Scope CurrentUser -Force
Unblock-File .\1-Generic_DevDrive_Setup.ps1

--- DISMOUNT EXISTING DEV DRIVE ---
-- Close all applications using the drive before dismounting
Dismount-DiskImage -ImagePath "C:\DevDrives\DEV_GENERIC.vhdx"
Dismount-DiskImage -ImagePath "H:\DevDrives\DEV_DEGORAS_SLR.vhdx"

--- DELETE VIRTUAL DISK FILE (IF RECREATING IT) ---
Remove-Item -Path "C:\DevDrives\DEV_GENERIC.vhdx"
Remove-Item -Path "H:\DevDrives\DEV_DEGORAS_SLR.vhdx"

--- MANUALLY MOUNT AN EXISTING DEV DRIVE ---
Mount-DiskImage -ImagePath "C:\DevDrives\DEGORAS_DEV.vhdx"

--- RESIZE THE VIRTUAL DISK ---
Resize-VHD -Path "C:\DevDrives\DEGORAS_DEV.vhdx" -SizeBytes 40GB

--- CHECK CURRENT DISK SIZE ---
Get-VHD -Path "C:\DevDrives\DEGORAS_DEV.vhdx" | Select Path, FileSize, Size
